# The dialogs' fonts (resources/fonts: Barlow, SIL Open Font License 1.1), embedded as byte arrays
# (ui_fonts.h). The file is rewritten only when a font changes.
set(REXUI_FONTS
    "kBarlowRegular:Barlow-Regular.ttf"
    "kBarlowSemiBold:Barlow-SemiBold.ttf"
    "kBarlowCondensedBold:BarlowCondensed-Bold.ttf"
)
set(REXUI_FONTS_CPP "${CMAKE_CURRENT_BINARY_DIR}/ui_fonts_data.cpp")
# Built with string(APPEND) only: the C++ has semicolons, which a CMake list would split on.
set(_rexui_fonts_content "// Generated by src/ui/CMakeLists.txt from resources/fonts - DO NOT EDIT\n\n")
string(APPEND _rexui_fonts_content "#include \"ui/ui_fonts.h\"\n\nnamespace rex::ui::fonts {\n")
foreach(_rexui_font IN LISTS REXUI_FONTS)
    string(REPLACE ":" ";" _rexui_font_parts "${_rexui_font}")
    list(GET _rexui_font_parts 0 _rexui_font_name)
    list(GET _rexui_font_parts 1 _rexui_font_file)
    set(_rexui_font_path "${REXGLUE_ROOT}/resources/fonts/${_rexui_font_file}")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_rexui_font_path}")
    file(READ "${_rexui_font_path}" _rexui_font_hex HEX)
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," _rexui_font_bytes "${_rexui_font_hex}")
    string(LENGTH "${_rexui_font_hex}" _rexui_font_hex_length)
    math(EXPR _rexui_font_size "${_rexui_font_hex_length} / 2")
    string(APPEND _rexui_fonts_content "const unsigned char ${_rexui_font_name}[] = {${_rexui_font_bytes}};\n")
    string(APPEND _rexui_fonts_content "const size_t ${_rexui_font_name}Size = ${_rexui_font_size};\n")
endforeach()
string(APPEND _rexui_fonts_content "}  // namespace rex::ui::fonts\n")
file(WRITE "${REXUI_FONTS_CPP}.tmp" "${_rexui_fonts_content}")
configure_file("${REXUI_FONTS_CPP}.tmp" "${REXUI_FONTS_CPP}" COPYONLY)
list(APPEND REXUI_SOURCES "${REXUI_FONTS_CPP}")
